cmake_minimum_required(VERSION 3.20)
project(CouchSplit VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX SDL_MAIN_HANDLED)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network)
if(WIN32)
    # SDL's pkg-config target adds -Dmain=SDL_main, which replaces Qt's Unicode entry point.
    find_package(SDL2 CONFIG REQUIRED)
    set(COOP_SDL_TARGET SDL2::SDL2)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    set(COOP_SDL_TARGET PkgConfig::SDL2)
endif()
option(COUCHSPLIT_GBA "Build the GBA worker" ON)
set(COUCHSPLIT_MGBA_SOURCE "" CACHE PATH "Pinned mGBA 0.10.5 source (optional offline build)")
if(COUCHSPLIT_GBA)
    add_subdirectory(gba)
endif()
qt_add_executable(platinum-coop launcher/main.cpp)
if(WIN32)
    set_target_properties(platinum-coop PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
target_include_directories(platinum-coop PRIVATE controller display saves launcher)
target_link_libraries(platinum-coop PRIVATE Qt6::Widgets Qt6::Network ${COOP_SDL_TARGET})
target_compile_options(platinum-coop PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(coop-tests tests/test_core.cpp)
    target_include_directories(coop-tests PRIVATE controller display saves launcher)
    target_link_libraries(coop-tests PRIVATE Qt6::Widgets Qt6::Network ${COOP_SDL_TARGET})
    target_compile_options(coop-tests PRIVATE -UNDEBUG)
    add_executable(worker-fixture tests/worker_fixture.cpp)
    target_include_directories(worker-fixture PRIVATE display)
    target_link_libraries(worker-fixture PRIVATE Qt6::Core)
    add_test(NAME core COMMAND coop-tests $<TARGET_FILE:worker-fixture>)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;SDL_AUDIODRIVER=dummy")
    add_executable(launcher-tests tests/test_launcher.cpp)
    target_include_directories(launcher-tests PRIVATE controller display saves launcher)
    target_link_libraries(launcher-tests PRIVATE Qt6::Widgets Qt6::Network ${COOP_SDL_TARGET})
    target_compile_options(launcher-tests PRIVATE -UNDEBUG)
    add_test(NAME launcher-ui COMMAND launcher-tests $<TARGET_FILE:worker-fixture>)
    set_tests_properties(launcher-ui PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;SDL_AUDIODRIVER=dummy")
    if(COUCHSPLIT_GBA)
        foreach(firered_component pokemon world connections inventory storage rewards dex hooks capture bench party_menu progress personal statistics encounter campaign campaign_runtime campaign_sync audio)
            add_executable(firered-${firered_component}-tests tests/test_firered_${firered_component}.cpp)
            target_compile_definitions(firered-${firered_component}-tests PRIVATE M_CORE_GBA)
            target_link_libraries(firered-${firered_component}-tests PRIVATE couchsplit-mgba-core)
            target_compile_options(firered-${firered_component}-tests PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
            add_test(NAME firered-${firered_component} COMMAND firered-${firered_component}-tests)
        endforeach()
        add_executable(verify-gba tests/verify_gba.cpp)
        target_include_directories(verify-gba PRIVATE controller display saves launcher)
        target_link_libraries(verify-gba PRIVATE Qt6::Widgets Qt6::Network ${COOP_SDL_TARGET})
        # Explicit, ROM-dependent experiment. Neither a release binary nor a CTest
        # fixture: requires a privately created fresh-intro checkpoint.
        add_executable(verify-firered-team-input EXCLUDE_FROM_ALL tests/verify_firered_team_input.cpp)
        target_compile_definitions(verify-firered-team-input PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-team-input PRIVATE couchsplit-mgba-core Qt6::Gui)
        add_executable(verify-firered-pair EXCLUDE_FROM_ALL tests/verify_firered_pair.cpp)
        target_compile_definitions(verify-firered-pair PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-pair PRIVATE couchsplit-mgba-core Qt6::Gui)
        add_executable(verify-firered-story EXCLUDE_FROM_ALL tests/verify_firered_story.cpp)
        target_compile_definitions(verify-firered-story PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-story PRIVATE couchsplit-mgba-core Qt6::Gui)
        add_executable(verify-firered-campaign EXCLUDE_FROM_ALL tests/verify_firered_campaign.cpp)
        target_compile_definitions(verify-firered-campaign PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-campaign PRIVATE couchsplit-mgba-core Qt6::Gui)
        add_executable(verify-firered-gyms EXCLUDE_FROM_ALL tests/verify_firered_gyms.cpp)
        target_compile_definitions(verify-firered-gyms PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-gyms PRIVATE couchsplit-mgba-core Qt6::Gui)
        add_executable(verify-firered-progression EXCLUDE_FROM_ALL tests/verify_firered_progression.cpp)
        target_compile_definitions(verify-firered-progression PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-progression PRIVATE couchsplit-mgba-core Qt6::Gui)
        add_executable(verify-firered-connections EXCLUDE_FROM_ALL tests/verify_firered_connections.cpp)
        target_compile_definitions(verify-firered-connections PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-connections PRIVATE couchsplit-mgba-core Qt6::Gui)
        add_executable(verify-firered-map-setup EXCLUDE_FROM_ALL tests/verify_firered_map_setup.cpp)
        target_compile_definitions(verify-firered-map-setup PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-map-setup PRIVATE couchsplit-mgba-core Qt6::Core)
        add_executable(verify-firered-public-saves EXCLUDE_FROM_ALL tests/verify_firered_public_saves.cpp)
        target_compile_definitions(verify-firered-public-saves PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-public-saves PRIVATE couchsplit-mgba-core Qt6::Core)
        add_executable(verify-firered-public-pair EXCLUDE_FROM_ALL tests/verify_firered_public_pair.cpp)
        target_compile_definitions(verify-firered-public-pair PRIVATE M_CORE_GBA)
        target_link_libraries(verify-firered-public-pair PRIVATE couchsplit-mgba-core Qt6::Core)
        add_executable(create-firered-checkpoint EXCLUDE_FROM_ALL tests/create_firered_checkpoint.cpp)
        target_compile_definitions(create-firered-checkpoint PRIVATE M_CORE_GBA)
        target_link_libraries(create-firered-checkpoint PRIVATE couchsplit-mgba-core Qt6::Core)
    endif()
endif()
# Embed the licensed pixel heading font in both production and native UI tests.
foreach(ui_target platinum-coop launcher-tests verify-gba)
    if(TARGET ${ui_target})
        qt_add_resources(${ui_target} "launcher_assets"
            PREFIX "/launcher" BASE "assets"
            FILES assets/fonts/Silkscreen-Regular.ttf)
    endif()
endforeach()
install(TARGETS platinum-coop RUNTIME DESTINATION bin)
